# ryo-kit Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the ryo-kit CLI and skill templates — a meta-framework that generates custom AI-driven development frameworks tailored to an org's context.

**Architecture:** Deterministic CLI (Commander.js + @clack/prompts) that collects org context via TUI interview, auto-detects project artifacts, and installs bootstrapping skills into 6 AI runtimes. The installed skills (markdown prompt files) run in the user's AI tool to generate agents, skills, processes, and workflows. File-based persistence in `.ryo/.state/` enables cross-session resume.

**Tech Stack:** Node.js 20+ ESM, Commander.js, @clack/prompts, Zod, yaml, node:test

**Spec:** `ryo-kit-project-spec.md` in the repo root

```javascript
import { join } from 'node:path';
import { exists } from '../utils/fs.js';

const ARTIFACT_MAP = {
  'CLAUDE.md': { runtime: 'claude-code' },
  '.claude': { runtime: 'claude-code' },
  '.cursorrules': { runtime: 'cursor' },
  '.cursor': { runtime: 'cursor' },
  '.github/copilot-instructions.md': { runtime: 'copilot' },
  '.github/prompts': { runtime: 'copilot' },
  'AGENTS.md': { runtime: 'codex' },
  '.windsurfrules': { runtime: 'windsurf' },
  'GEMINI.md': { runtime: 'gemini-cli' },
  '.gemini': { runtime: 'gemini-cli' },
};

const LANGUAGE_FILES = {
  'package.json': 'javascript',
  'tsconfig.json': 'typescript',
  'requirements.txt': 'python',
  'pyproject.toml': 'python',
  'go.mod': 'go',
  'Cargo.toml': 'rust',
  'pom.xml': 'java',
  'build.gradle': 'java',
  'Gemfile': 'ruby',
  '*.csproj': 'csharp',
  '*.sln': 'csharp',
};

export async function detectProjectContext(projectDir) {
  const existing = [];
  const runtimes = new Set();
  const languages = new Set();

  // Check artifact files
  for (const [file, meta] of Object.entries(ARTIFACT_MAP)) {
    if (await exists(join(projectDir, file))) {
      existing.push(file);
      if (meta.runtime) runtimes.add(meta.runtime);
    }
  }

  // Check language files (exact match)
  for (const [file, lang] of Object.entries(LANGUAGE_FILES)) {
    if (file.startsWith('*')) continue; // handled below
    if (await exists(join(projectDir, file))) {
      languages.add(lang);
    }
  }

  // Check language files (glob patterns via readdir)
  const { readdir } = await import('node:fs/promises');
  try {
    const entries = await readdir(projectDir);
    for (const [pattern, lang] of Object.entries(LANGUAGE_FILES)) {
      if (!pattern.startsWith('*')) continue;
      const ext = pattern.slice(1); // e.g. '.csproj'
      if (entries.some(e => e.endsWith(ext))) {
        languages.add(lang);
      }
    }
  } catch { /* directory may not be readable */ }

  return {
    existing,
    runtimes: [...runtimes],
    languages: [...languages],
  };
}
```

```javascript
import { join } from 'node:path';
import { readFile, writeFile, copyFile } from 'node:fs/promises';
import { fileURLToPath } from 'node:url';
import { dirname } from 'node:path';
import { writeYaml } from '../utils/yaml.js';
import { ensureDir } from '../utils/fs.js';
import { OrgContextSchema } from './schema.js';

const __dirname = dirname(fileURLToPath(import.meta.url));
const TEMPLATES_DIR = join(__dirname, '..', '..', 'templates', 'defaults');

export async function writeOrgContext(targetDir, context) {
  // Validate before writing
  OrgContextSchema.parse(context);
  await ensureDir(targetDir);
  await writeYaml(join(targetDir, 'org-context.yaml'), context);

export async function writeConstitution(targetDir) {
  await ensureDir(targetDir);
  const template = await readFile(join(TEMPLATES_DIR, 'constitution.md'), 'utf8');
  await writeFile(join(targetDir, 'constitution.md'), template, 'utf8');
}

export async function writeDefaultTemplates(targetDir) {
  const templatesDir = join(targetDir, 'templates');
  await ensureDir(templatesDir);
  await copyFile(
    join(TEMPLATES_DIR, 'agent-base.yaml'),
    join(templatesDir, 'agent-base.yaml')
  );
  await copyFile(
    join(TEMPLATES_DIR, 'process-base.yaml'),
    join(templatesDir, 'process-base.yaml')

}
```

```javascript
import { join } from 'node:path';
import { readFile, writeFile, rm } from 'node:fs/promises';
import { BaseRuntime, RYO_BLOCK_START, RYO_BLOCK_END } from './base.js';
import { ensureDir, readIfExists } from '../utils/fs.js';

export class ClaudeCodeRuntime extends BaseRuntime {
  get name() { return 'claude-code'; }
  get skillsDir() { return join(this.projectDir, '.claude', 'skills'); }
  get configFile() { return join(this.projectDir, 'CLAUDE.md'); }

  async installSkill(skillName, skillContent) {
    const skillDir = join(this.skillsDir, skillName);
    await ensureDir(skillDir);
    await writeFile(join(skillDir, 'SKILL.md'), skillContent, 'utf8');
  }

  async updateConfig(contextRef) {
    const block = [
      RYO_BLOCK_START,
      '',
      '## ryo-kit Framework',
      '',
      `This project uses ryo-kit. Framework files are in \`${contextRef}\`.`,
      'See `.ryo/process.md` for the development process definition.',
      'See `.ryo/agents/` for agent definitions and `.ryo/skills/` for skill definitions.',
      '',
      RYO_BLOCK_END,
    ].join('\n');

    const existing = await readIfExists(this.configFile);
    if (existing) {
      // Remove old block if present, then append
      const cleaned = removeRyoBlock(existing);
      await writeFile(this.configFile, cleaned.trimEnd() + '\n\n' + block + '\n', 'utf8');
    } else {
      await writeFile(this.configFile, block + '\n', 'utf8');
    }
  }

  async uninstall() {
    // Remove ryo skill directories
    const { readdir } = await import('node:fs/promises');
    try {
      const entries = await readdir(this.skillsDir);
      for (const entry of entries) {
        if (entry.startsWith('ryo-')) {
          await rm(join(this.skillsDir, entry), { recursive: true });
        }
      }
    } catch { /* skills dir may not exist */ }

    // Remove ryo block from config
    const existing = await readIfExists(this.configFile);
    if (existing) {
      const cleaned = removeRyoBlock(existing);
      await writeFile(this.configFile, cleaned, 'utf8');
    }
  }
}

function removeRyoBlock(content) {
  const startIdx = content.indexOf(RYO_BLOCK_START);
  const endIdx = content.indexOf(RYO_BLOCK_END);
  if (startIdx === -1 || endIdx === -1) return content;
  return (
    content.slice(0, startIdx) +
    content.slice(endIdx + RYO_BLOCK_END.length)
  ).replace(/\n{3,}/g, '\n\n');
}
```
